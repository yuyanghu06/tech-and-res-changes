# Centerpieces for Russian-pack Eastern Europe city variants. Root = state region.
# Hand-maintained companion to city_types/ztr_eeu_russian_city.txt.
# Historical stages use the Russian pack; modern stages keep their technology-selected custom models.
# Skyscraper centerpieces require an existing skyscraper that is being built or has a completed level.

ztr_eeu_russian_cp_city_centerpiece_01 = {
	city_graphics_type = ztr_eeu_russian_city_early_modern_start
	city_graphics_type = ztr_eeu_russian_city_early_modern
	city_graphics_type = ztr_eeu_russian_city_early_modern_socialist
	grid_size = { 2 2 }
	attach = { variant = { entity = russian_city_centerpiece_01_entity } }
	attach = { variant = { entity = russian_city_centerpiece_02_entity } }
}

ztr_eeu_russian_cp_city_centerpiece_01_skyscraper = {
	city_graphics_type = ztr_eeu_russian_city_early_modern_start
	city_graphics_type = ztr_eeu_russian_city_early_modern
	city_graphics_type = ztr_eeu_russian_city_early_modern_socialist
	priority = 10
	grid_size = { 2 2 }

	trigger = {
		any_scope_state = {
			any_scope_building = {
				is_building_type = building_skyscraper
				OR = {
					is_under_construction = yes
					level > 0
				}
			}
		}
	}

	attach = { variant = { entity = russian_city_centerpiece_01_entity 	building_type = building_skyscraper } }
	attach = { variant = { entity = russian_city_skyscraper_01_entity 	building_type = building_skyscraper } }
}
